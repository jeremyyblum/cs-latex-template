% Glossar-Einträge
% Syntax: \newglossaryentry{key}{name={Term}, description={Definition}}

\newglossaryentry{ml}{
    name={Machine Learning},
    description={The automated detection of meaningful patterns in data, enabling systems to improve performance on a specific task through experience}
}

\newglossaryentry{supervisedlearning}{
    name={Supervised Learning},
    description={A machine learning paradigm where algorithms learn from labeled training data (input-output pairs) to make predictions on new, unseen data}
}

\newglossaryentry{binaryclassification}{
    name={Binary Classification},
    description={A supervised learning task where the goal is to classify instances into one of two possible classes (e.g., spam vs. not spam)}
}

\newglossaryentry{boosting}{
    name={Boosting},
    description={An ensemble method that sequentially combines multiple weak learners to create a strong predictive model, with each new model focusing on correcting errors of previous ones}
}

\newglossaryentry{gradientboosting}{
    name={Gradient Boosting},
    description={A boosting technique that frames the ensemble learning process as gradient descent optimization in function space}
}

\newglossaryentry{xgboost}{
    name={XGBoost},
    description={Extreme Gradient Boosting - a scalable machine learning system for tree boosting with optimized performance and regularization}
}

\newglossaryentry{overfitting}{
    name={Overfitting},
    description={The production of an analysis which corresponds too closely to training data and may fail to generalize to new data or predict future observations reliably}
}

\newglossaryentry{crossvalidation}{
    name={Cross-Validation},
    description={A statistical method for evaluating machine learning models by partitioning data into training and validation sets multiple times}
}

\newglossaryentry{precision}{
    name={Precision},
    description={The proportion of positive predictions that are actually correct (True Positives / (True Positives + False Positives))}
}

\newglossaryentry{recall}{
    name={Recall},
    description={The proportion of actual positive cases that are correctly identified (True Positives / (True Positives + False Negatives))}
}

\newglossaryentry{f1score}{
    name={F1-Score},
    description={The harmonic mean of precision and recall, providing a balanced measure of classifier performance}
}

\newglossaryentry{aucroc}{
    name={AUC-ROC},
    description={Area Under the Receiver Operating Characteristic Curve - a performance metric that measures the quality of predictions across all classification thresholds}
}
